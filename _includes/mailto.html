{%- comment -%} Prints a prefilled mailto: URL (for an href) from a contact data file. Usage: {% include mailto.html contact="ai_contact" plan="Audit" %}; leave out `plan` for a general enquiry. Callers add target="_blank" so the email opens in a new tab. uri_escape turns spaces into %20 and line breaks into %0A; the replace chain then encodes the characters uri_escape leaves alone but that would break a mailto link (& starts a new field, # ends the link, ? and + can be misread). url_encode is not used because it turns spaces into "+". {%- endcomment -%} {%- assign c = site.data[include.contact] -%} {%- assign subject_plan = include.plan | default: c.general_plan -%} {%- assign body_plan = include.plan | default: "" -%} {%- capture subject -%}{{ c.subject | replace: "[plan]", subject_plan }}{%- endcapture -%} {%- capture body -%}{{ c.body | replace: "[plan]", body_plan | strip }}{%- endcapture -%} {%- capture subject_q -%}{{ subject | uri_escape | replace: "&", "%26" | replace: "?", "%3F" | replace: "#", "%23" | replace: "+", "%2B" }}{%- endcapture -%} {%- capture body_q -%}{{ body | uri_escape | replace: "&", "%26" | replace: "?", "%3F" | replace: "#", "%23" | replace: "+", "%2B" }}{%- endcapture -%} {%- capture href -%}mailto:{{ c.email }}?subject={{ subject_q }}&body={{ body_q }}{%- endcapture -%}{{ href }}
